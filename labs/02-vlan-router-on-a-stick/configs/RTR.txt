! RTR (Cisco ISR4331) - router-on-a-stick + DHCP for both VLANs
enable
configure terminal
hostname RTR
!
! Physical interface carries the 802.1Q trunk; it needs no IP of its own
interface GigabitEthernet0/0/0
 description Trunk to SW
 no shutdown
 exit
!
! One subinterface per VLAN = that VLAN's default gateway
interface GigabitEthernet0/0/0.10
 description VLAN 10 SALES gateway
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 exit
interface GigabitEthernet0/0/0.20
 description VLAN 20 IT gateway
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 exit
!
! DHCP: keep .1-.9 for gateways/servers, hand out the rest
ip dhcp excluded-address 192.168.10.1 192.168.10.9
ip dhcp excluded-address 192.168.20.1 192.168.20.9
ip dhcp pool SALES
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
 exit
ip dhcp pool IT
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 8.8.8.8
 exit
end
write memory
